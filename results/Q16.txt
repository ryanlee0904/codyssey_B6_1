-- [Q16] 수정 | 결제 완료(PAID)된 주문의 수량만큼 상품 재고를 차감한다 (UPDATE + 상관 서브쿼리)

SELECT product_id, name, stock AS stock_before
FROM product
WHERE product_id IN (SELECT oi.product_id
                     FROM order_item oi
                     INNER JOIN orders o ON o.order_id = oi.order_id
                     WHERE o.status = 'PAID')
ORDER BY product_id;

UPDATE product
SET stock = stock - (
        SELECT SUM(oi.quantity)
        FROM order_item oi
        INNER JOIN orders o ON o.order_id = oi.order_id
        WHERE o.status = 'PAID'
          AND oi.product_id = product.product_id
    )
WHERE product_id IN (
        SELECT oi.product_id
        FROM order_item oi
        INNER JOIN orders o ON o.order_id = oi.order_id
        WHERE o.status = 'PAID'
    );

SELECT product_id, name, stock AS stock_after
FROM product
WHERE product_id IN (SELECT oi.product_id
                     FROM order_item oi
                     INNER JOIN orders o ON o.order_id = oi.order_id
                     WHERE o.status = 'PAID')
ORDER BY product_id;
