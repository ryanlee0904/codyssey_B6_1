-- [Q13] 서브쿼리 | 전체 상품 평균 가격보다 비싼 상품을 조회한다 (WHERE 절 스칼라 서브쿼리)

SELECT product_id, name, price,
       (SELECT ROUND(AVG(price), 0) FROM product) AS overall_avg_price
FROM product
WHERE price > (SELECT AVG(price) FROM product)
ORDER BY price DESC;
